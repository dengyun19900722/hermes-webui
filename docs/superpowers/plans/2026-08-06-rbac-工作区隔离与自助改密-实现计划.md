```python
def update_password(state_dir: Path, user_id: str, new_password_hash: str) -> dict[str, Any]:
    """Replace the user's password_hash. Raises KeyError if user not found."""
    users = load_users(state_dir)
    for user in users:
        if user.get("id") == user_id:
            user["password_hash"] = new_password_hash
            save_users(state_dir, users)
            return user
    raise KeyError(f"User not found: {user_id}")


def _hash_password_for_test(plain: str) -> str:
    """Deterministic PBKDF2 hash for tests; not for production use."""
    import hashlib
    salt = b"test-salt-fixed-for-reproducibility"
    return "pbkdf2_sha256$" + hashlib.pbkdf2_hmac(
        "sha256", plain.encode("utf-8"), salt, 1000
    ).hex()
```

注意：`_hash_password_for_test` 用 `hashlib.pbkdf2_hmac` 直接生成；生产用 `api.auth._hash_password`。本 helper **仅供测试**。

```python
def invalidate_all_user_sessions(user_id: str, *, keep_token: str | None = None) -> int:
    """Invalidate all RBAC sessions for user_id. Returns count removed.
    
    If keep_token is provided, that token is preserved (so the user is not
    kicked out of their own current session after a self-password-change).
    """
    sessions_path = _STATE_DIR / ".rbac-sessions.json"
    if not sessions_path.exists():
        return 0
    try:
        data = json.loads(sessions_path.read_text(encoding="utf-8"))
    except Exception:
        return 0
    if not isinstance(data, dict):
        return 0
    removed = 0
    new_data = {}
    for token, info in data.items():
        if keep_token and token == keep_token:
            new_data[token] = info
            continue
        if isinstance(info, dict) and info.get("user_id") == user_id:
            removed += 1
            continue
        new_data[token] = info
    if removed:
        try:
            tmp = sessions_path.with_suffix(".json.tmp")
            tmp.write_text(json.dumps(new_data, ensure_ascii=False), encoding="utf-8")
            tmp.replace(sessions_path)
        except Exception:
            logger.exception("Failed to persist invalidated sessions")
    return removed
```

### 2.3 实现 `handle_auth_change_password`

```python
import re

def _validate_password_complexity(password: str) -> str | None:
    """Return error message or None if password meets rules:
    ≥ 8 chars, contains letters AND digits.
    """
    if len(password) < 8:
        return "password_too_short"
    if not re.search(r"[A-Za-z]", password) or not re.search(r"\d", password):
        return "password_needs_classes"
    return None


def handle_auth_change_password(handler, parsed) -> bool:
    """POST /api/auth/change-password — user changes own password.
    Body: {old_password, new_password}. Verifies old, validates complexity,
    updates hash, invalidates all OTHER sessions (keeps current).
    """
    from api.auth import (
        _hash_password as auth_hash_password,
        invalidate_all_user_sessions,
    )
    from api.user_store import find_user_by_id, update_password
    
    body = _read_json_body(handler)
    old_pw = (body.get("old_password") or "").strip()
    new_pw = (body.get("new_password") or "").strip()
    if not old_pw or not new_pw:
        _send_json(handler, 400, {"error": "old_password and new_password required"})
        return True
    
    user = _current_user(handler)
    if not user:
        _send_json(handler, 401, {"error": "Not authenticated"})
        return True
    
    # Verify old password
    state_dir = _get_state_dir()
    fresh = find_user_by_id(state_dir, user["id"])
    if not fresh:
        _send_json(handler, 401, {"error": "User not found"})
        return True
    
    if not verify_password(old_pw, fresh["password_hash"]):
        _send_json(handler, 400, {"error": "password_old_wrong"})
        return True
    
    err = _validate_password_complexity(new_pw)
    if err:
        _send_json(handler, 400, {"error": err})
        return True
    
    # Update
    new_hash = auth_hash_password(new_pw)
    update_password(state_dir, user["id"], new_hash)
    
    # Invalidate other sessions, keep current
    current_token = _get_session_token(handler)
    invalidate_all_user_sessions(user["id"], keep_token=current_token)
    
    _audit.write(
        category="rbac",
        action="password.change",
        actor_id=user["id"],
        actor_name=user.get("username"),
    )
    _send_json(handler, 200, {"ok": True})
    return True


def _get_state_dir() -> Path:
    """Resolve state dir (single source of truth)."""
    from api.config import STATE_DIR
    return Path(STATE_DIR)
```

```python
def handle_admin_reset_password(handler, parsed, user_id: str) -> bool:
    """PUT /api/admin/users/{user_id}/password — admin resets user password."""
    from api.auth import _hash_password as auth_hash_password, invalidate_all_user_sessions
    from api.user_store import find_user_by_id, update_password
    
    caller = _current_user(handler)
    if not caller:
        _send_json(handler, 401, {"error": "Not authenticated"})
        return True
    if not is_admin(caller):
        _send_json(handler, 403, {"error": "Admin only"})
        return True
    
    state_dir = _get_state_dir()
    target = find_user_by_id(state_dir, user_id)
    if not target:
        _send_json(handler, 404, {"error": "User not found"})
        return True
    
    body = _read_json_body(handler)
    new_pw = (body.get("new_password") or "").strip()
    if not new_pw:
        _send_json(handler, 400, {"error": "new_password required"})
        return True
    # Admin reset BYPASSES complexity check by design
    
    new_hash = auth_hash_password(new_pw)
    update_password(state_dir, user_id, new_hash)
    # Don't keep admin's current token (admin session unrelated)
    invalidate_all_user_sessions(user_id)
    
    _audit.write(
        category="rbac",
        action="password.reset",
        actor_id=caller["id"],
        actor_name=caller.get("username"),
        target_id=user_id,
        target_name=target.get("username"),
    )
    _send_json(handler, 200, {"ok": True})
    return True
```

```python
class WorkspacePermissionError(PermissionError):
    """Raised when a user attempts a workspace op without owner/admin role."""


def visible_workspaces(user_id: str, is_admin: bool, all_ws: list[dict]) -> list[dict]:
    """Filter workspace list by RBAC visibility."""
    if is_admin:
        return all_ws
    return [w for w in all_ws
            if user_id == w.get("owner")
            or user_id in (w.get("members") or [])]


def _require_workspace_op(ws: dict, user: dict, op: str) -> None:
    """Raise WorkspacePermissionError if user is neither admin nor owner."""
    is_admin = (user.get("role") or "") == "admin"
    if is_admin:
        return
    if ws.get("owner") == user.get("id"):
        return
    raise WorkspacePermissionError(
        f"Only owner or admin can {op} workspace"
    )


def _migrate_workspace_access(workspaces: list, state_dir: Path) -> list:
    """Idempotent: backfill owner/members for workspaces missing these fields."""
    from api.user_store import load_users
    
    users = load_users(state_dir)
    admins = sorted(
        [u for u in users if u.get("role") == "admin"],
        key=lambda u: u.get("created_at") or "",
    )
    if not admins:
        logger.warning("RBAC workspace migration skipped: no admin user found")
        return workspaces
    
    first_admin_id = admins[0]["id"]
    needs_persist = False
    for w in workspaces:
        if "owner" not in w or "members" not in w:
            w["owner"] = first_admin_id
            w["members"] = [first_admin_id]
            needs_persist = True
    
    if needs_persist:
        save_workspaces(workspaces)
        try:
            from api.audit import write as audit_write
            audit_write(
                category="rbac",
                action="workspace.migration",
                owner_default=first_admin_id,
                count=sum(1 for w in workspaces if w.get("owner") == first_admin_id),
            )
        except Exception:
            logger.debug("audit write failed for workspace.migration", exc_info=True)
    
    return workspaces
```

```python
    with socket.socket(socket.AF_INET, socket.SOCK_STREAM) as s:

        return s.getsockname()[1]


def _post(url, data, cookie=None):
    body = json.dumps(data).encode('utf-8')
    req = urllib.request.Request(url, data=body, method='POST',
                                 headers={'Content-Type': 'application/json'})
    if cookie:
        req.add_header('Cookie', cookie)
    try:
        with urllib.request.urlopen(req) as r:
            return r.status, dict(r.headers), r.read().decode('utf-8')
    except urllib.error.HTTPError as e:
        return e.code, dict(e.headers), e.read().decode('utf-8')


def _get(url, cookie=None):

    if cookie:
        req.add_header('Cookie', cookie)

    except urllib.error.HTTPError as e:
        return e.code, e.read().decode('utf-8')
```
